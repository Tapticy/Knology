
# Ternary Conditional 
```c#
      int[] draw1 = [24, 43, 45, 47, 49];
      int[] pick1 = new int[5];
      foreach (int d in draw1) {
          Console.Write($"\t {d}");
      }
      
      Console.WriteLine();
      string user = Convert.ToString(pick1);
      string sel = Convert.ToString(draw1);
      
 
      string total = (user == sel) ? "Match" : "No Match";
      Console.WriteLine(total);

      int input = new Random().Next(-5, 5);

      string classify;
      // Standard if statement
      if (input >= 0) {
          classify = "nonnegative";
      }
      else {
          classify = "negative";
      }
      Console.WriteLine(classify);
 
     // Ternary Conditional
      classify = (input >= 0) ? "This" : "That";
      Console.WriteLine(classify);
```
